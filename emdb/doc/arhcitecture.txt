package com.example.catalog.application;

import com.example.catalog.domain.CreditId;
import com.example.catalog.domain.Movie;
import com.example.catalog.domain.MovieId;
import com.example.catalog.domain.Movies;

import jakarta.enterprise.context.ApplicationScoped;
import jakarta.transaction.Transactional;

import java.util.UUID;

/**
 * Application service: one method per use case.
 *
 * It orchestrates — load aggregate, invoke domain behaviour, save aggregate — and owns
 * the transaction boundary. Business rules live in the {@link Movie} aggregate; SQL
 * details live behind the {@link Movies} port. The injected StatelessSession-backed
 * adapter enlists in the JTA transaction opened by {@code @Transactional}, so every
 * use case below is atomic.
 */
@ApplicationScoped
public class MovieCatalogService {

    private final Movies movies;

    public MovieCatalogService(Movies movies) {
        this.movies = movies;
    }

    /** Take a movie + credits you already fetched from TMDB and store them as one aggregate. */
    @Transactional
    public MovieId importFromTmdb(ImportMovieFromTmdb command) {
        MovieId id = new MovieId(command.tmdbMovieId());
        if (movies.exists(id)) {
            throw new IllegalStateException("Movie %d is already in the catalog".formatted(id.value()));
        }

        Movie movie = Movie.importFromTmdb(id, command.title(), command.releaseDate());
        command.cast().forEach(c -> movie.addCastMember(c.tmdbPersonId(), c.name(), c.character(), c.order()));
        command.crew().forEach(c -> movie.addCrewMember(c.tmdbPersonId(), c.name(), c.department(), c.job()));

        movies.add(movie);
        return movie.id();
    }

    @Transactional
    public Movie movie(long tmdbMovieId) {
        return requireMovie(tmdbMovieId);
    }

    /** UI edit: each credit has its own CreditId, so a single line can be addressed and fixed. */
    @Transactional
    public void renameCharacter(long tmdbMovieId, UUID creditId, String newCharacterName) {
        Movie movie = requireMovie(tmdbMovieId);
        movie.renameCharacter(new CreditId(creditId), newCharacterName);
        movies.update(movie); // stateless session = no dirty checking, so saving is explicit
    }

    @Transactional
    public void renamePerson(long tmdbMovieId, UUID creditId, String newName) {
        Movie movie = requireMovie(tmdbMovieId);
        movie.renamePerson(new CreditId(creditId), newName);
        movies.update(movie);
    }

    @Transactional
    public void removeCredit(long tmdbMovieId, UUID creditId) {
        Movie movie = requireMovie(tmdbMovieId);
        movie.removeCredit(new CreditId(creditId));
        movies.update(movie);
    }

    @Transactional
    public void deleteMovie(long tmdbMovieId) {
        MovieId id = new MovieId(tmdbMovieId);
        if (!movies.exists(id)) {
            throw new MovieNotFoundException(tmdbMovieId);
        }
        movies.remove(id);
    }

    private Movie requireMovie(long tmdbMovieId) {
        return movies.findById(new MovieId(tmdbMovieId))
                .orElseThrow(() -> new MovieNotFoundException(tmdbMovieId));
    }
}
